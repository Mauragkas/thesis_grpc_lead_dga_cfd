        $$BR = \frac{N_{\text{Tier 1}} + N_{\text{Tier 2}}}{N_{\text{total}}} = 1 - \frac{N_{\text{CFD}}}{N_{\text{total}}}$$

        \item \textbf{Μείωση κλήσεων VLM:} από 4.500 σε 1.972 ($-56.2\%$).
        \item \textbf{Wall-Clock Speedup: $2.28\times$} (6.0 min vs 13.8 min).
      \end{itemize}
